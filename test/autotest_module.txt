//==================================================================
//  Автотест (Автотест.ert) - модуль внешней обработки для тест-базы.
//  Авто-запуск: ПриОткрытии(). Компонента: КаталогИБ()+"1cmcp.dll".
//  Лог: КаталогИБ()+"test_log.txt". После запуска сервера пишет __READY__,
//  дальше сервер проверяет внешний клиент (test\mcp_client_test.py).
//  ВАЖНО (1С 7.7): НЕ использовать Возврат внутри Попытка.
//==================================================================
Перем глПутьЛога, глСервер;

Функция ПутьЛога()
	Если ПустаяСтрока(глПутьЛога)=1 Тогда
		глПутьЛога = КаталогИБ() + "test_log.txt";
	КонецЕсли;
	Возврат глПутьЛога;
КонецФункции

Процедура НачатьЛог()
	Попытка
		ФСО = СоздатьОбъект("Scripting.FileSystemObject");
		П = ФСО.CreateTextFile(ПутьЛога(), 1);
		П.Close();
	Исключение
	КонецПопытки;
КонецПроцедуры

Процедура ВЛог(Текст)
	Попытка
		ФСО = СоздатьОбъект("Scripting.FileSystemObject");
		П = ФСО.OpenTextFile(ПутьЛога(), 8, 1);
		П.WriteLine(Текст);
		П.Close();
	Исключение
	КонецПопытки;
КонецПроцедуры

Процедура Инстр(Имя, Арг)
	Попытка
		ВЛог(Имя + " " + Арг + " => " + глСервер.ВыполнитьИнструмент(Имя, Арг));
	Исключение
		ВЛог(Имя + " " + Арг + " => ИСКЛЮЧЕНИЕ " + ОписаниеОшибки());
	КонецПопытки;
КонецПроцедуры

Процедура ТестМСП()
	НачатьЛог();
	рз = 0;
	Попытка
		рз = ЗагрузитьВнешнююКомпоненту(КаталогИБ() + "1cmcp.dll");
	Исключение
		ВЛог("исключение при загрузке: " + ОписаниеОшибки());
	КонецПопытки;
	ВЛог("load=" + рз);
	Если рз <> 1 Тогда
		ВЛог("__DONE__");
		Возврат;
	КонецЕсли;
	Попытка
		глСервер = СоздатьОбъект("MCPСервер");
		ВЛог("version=" + глСервер.Версия());
	Исключение
		ВЛог("СоздатьОбъект: " + ОписаниеОшибки());
	КонецПопытки;
	Если ТипЗначенияСтр(глСервер) = "" Тогда
		ВЛог("__DONE__");
		Возврат;
	КонецЕсли;

	Сообщить("строка до теста");
	Инстр("eval", "{""expression"":""2+2""}");
	Инстр("eval", "{""expression"":""КаталогИБ()""}");
	Инстр("eval", "{""expression"":""ТекущаяДата()""}");
	Инстр("run_code", "{""code"":""Сообщить(\""привет из MCP\"");\nРезультат = 5*5;""}");
	Инстр("run_code", "{""code"":""Функция Кв(х)\n Возврат х*х;\nКонецФункции\nРезультат = Кв(7);""}");
	Инстр("run_code", "{""code"":""Результат = ;""}");
	Инстр("run_code", "{""code"":""а = 0;\nРезультат = 1/а;""}");
	Инстр("run_code", "{""code"":""Результат = НеизвестнаяФункция(1);""}");
	Инстр("run_code", "{""code"":""Сп = СоздатьОбъект(\""СписокЗначений\"");\nСп.ДобавитьЗначение(1);\nРезультат = Сп;""}");
	Инстр("read_messages", "{""max_lines"":10}");

	Попытка
		ок = глСервер.Запустить(18777, "test-token-123");
		ВЛог("start=" + ок + " url=" + глСервер.Адрес() + " err=" + глСервер.ПоследняяОшибка());
		ВЛог("token=" + глСервер.СгенерироватьТокен());
	Исключение
		ВЛог("Запустить: " + ОписаниеОшибки());
	КонецПопытки;
	ВЛог("__READY__");
КонецПроцедуры

Процедура ПриОткрытии()
	ТестМСП();
КонецПроцедуры
